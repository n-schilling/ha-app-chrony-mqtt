abi <abi/3.0>,
#include <tunables/global>

# chronyd serves NTP and may set the host clock (SYS_TIME); the web page and
# the sensor publisher read chronyc and talk to the MQTT broker. s6-overlay
# and the setup run as root; chronyd, the page and the publisher run as user
# chrony. Measured on Home Assistant OS in complain mode before it enforced.
profile chrony_ntp flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # s6-overlay and the setup prepare the folders of user chrony and start
  # the services as that user; chronyd binds port 123 before it drops root
  capability chown,
  capability dac_override,
  capability fowner,
  capability kill,
  capability net_bind_service,
  capability setgid,
  capability setuid,
  # set_system_clock
  capability sys_time,

  network inet dgram,
  network inet6 dgram,
  network inet stream,
  network inet6 stream,
  network unix,

  # s6 stops its services; the Supervisor stops the container
  signal (send, receive) peer=@{profile_name},
  signal (receive) peer=unconfined,

  # s6-overlay
  /init ix,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** rix,
  /run/** rwk,
  /run/ r,

  # Programs of the image
  /bin/** rix,
  /sbin/** rix,
  /usr/bin/** rix,
  /usr/sbin/** rix,
  /usr/lib/** rm,
  /usr/local/** rix,
  /lib/** rm,
  /usr/share/** r,
  /etc/** r,
  /etc/chrony/chrony.conf w,

  # bashio's options and API cache
  /tmp/ r,
  /tmp/** rwk,

  # The drift file, the page's clients and the options
  /data/ r,
  /data/** rwk,

  # The page dates chronyd's start; s6 and Python read their own process
  @{PROC}/sys/kernel/random/boot_id r,
  @{PROC}/@{pid}/** r,
  @{PROC}/ r,
  /dev/null rw,
  /dev/urandom r,
  /dev/random r,
  /dev/tty rw,
  /dev/pts/* rw,
  /dev/rtc0 r,

  deny /proc/sys/** w,
  deny /sys/** w,
}
